! ==============================================================================
! CONFIGURACION DE SWITCH 2 (SW2 - LAN 2)
! Modelo recomendado en Packet Tracer: Cisco Catalyst 2960
! ==============================================================================

enable
configure terminal
hostname SW2

! Configuracion IP para administracion remota (Opcional pero recomendada)
interface vlan 1
 description Interfaz de Gestion SW2
 ip address 192.168.20.2 255.255.255.0
 no shutdown
 exit

! Gateway predeterminado apuntando a R2
ip default-gateway 192.168.20.1

! Los puertos FastEthernet 0/1 al 0/24 en modo Access Vlan 1
interface range FastEthernet0/1 - 24
 switchport mode access
 switchport access vlan 1
 no shutdown
 exit

end
write memory
