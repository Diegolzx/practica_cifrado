! ==============================================================================
! CONFIGURACION DE ROUTER 2 (R2 - FISICO)
! ==============================================================================

enable
configure terminal
hostname R2
no ip domain-lookup

! 1. Interfaz hacia Switch 2 (LAN 2)
interface GigabitEthernet0/0
 description Conexion LAN 2 hacia Switch 2
 ip address 192.168.20.1 255.255.255.0
 no shutdown
 exit

! 2. Enlace WAN / NUBE hacia R1
! NOTA: Usa la opcion que tengan disponible en su laboratorio:

! ---> OPCION A: Si se conectan por CABLE SERIAL (DTE en R2):
interface Serial0/0/0
 description Enlace WAN hacia R1
 ip address 10.0.0.2 255.255.255.252
 no shutdown
 exit

! ---> OPCION B: Si se conectan por CABLE GIGABIT ETHERNET DIRECTO (RJ-45):
! interface GigabitEthernet0/1
!  description Enlace WAN hacia R1
!  ip address 10.0.0.2 255.255.255.252
!  no shutdown
!  exit

! 3. Enrutamiento hacia LAN 1 (Laptop 1)
ip route 192.168.10.0 255.255.255.0 10.0.0.1

end
write memory
